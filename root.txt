    def _start_status_polling(self):
        # 1. Download Queue အခြေအနေကို စစ်ဆေးမည်
        queue = self.db.fetch_one(
            "SELECT id, total_items, completed_count, status FROM download_queues ORDER BY id DESC LIMIT 1"
        )

        if queue and queue["status"] == "running":
            total = queue["total_items"] or 0
            completed = queue["completed_count"] or 0
            percent = (completed / total * 100) if total > 0 else 0.0

            # UI Progress Bar နှင့် Label ကို Update လုပ်မည်
            self.progress_bar["value"] = percent
            self.lbl_progress.config(
                text=f"Downloading Progress: {percent:.1f}% ({completed:,} / {total:,} files complete)"
            )
            self.lbl_status.config(text=f"Status: DOWNLOADING (Queue #{queue['id']})", foreground="green")

        else:
            # 2. Scan Progress (သို့) မူလ Sync Job Status ကို စစ်ဆေးမည်
            drive_id = self.ent_drive_id.get().strip()
            if drive_id:
                job = self.db.fetch_one(
                    "SELECT id, status, scanned_files, scanned_folders FROM sync_jobs WHERE drive_id = ? ORDER BY id DESC LIMIT 1",
                    (drive_id,)
                )
                if job:
                    colors = {"running": "green", "paused": "orange", "cancelled": "red", "completed": "blue", "failed": "red"}
                    self.lbl_status.config(text=f"Status: {job['status'].upper()}", foreground=colors.get(job["status"], "black"))
                    self.lbl_files.config(text=f"Scanned Files: {job['scanned_files']:,}")
                    self.lbl_folders.config(text=f"Scanned Folders: {job['scanned_folders']:,}")
                    self.lbl_job_id.config(text=f"Active Job ID: #{job['id']}")

                    # Scan သက်သက် လုပ်နေချိန်တွင် Progress Bar ကို Reset ထားမည်
                    if job["status"] != "running":
                        self.progress_bar["value"] = 0
                        self.lbl_progress.config(text="Progress: 0.0%")

        # စက္ကန့်ဝက် (500ms) တိုင်း အလိုအလျောက် ပြန်လည် စစ်ဆေးမည်
        self.after(500, self._start_status_polling)

သို့သော် download_items Table ထဲသို့မူ File များကိုသာ ခွဲထုတ်ထည့်ပေးရပါမည် (Folder များကို aria2c ဖြင့် ဒေါင်းလုဒ်ဆွဲရန် မလိုဘဲ Python ရဲ့ os.makedirs() ဖြင့် Local Disk ပေါ်တွင် အလိုအလျောက် တည်ဆောက်သွားမည် ဖြစ်သည်)။

အဆင့် (၁) - drive_files Table တွင် File/Folder ခွဲခြားသိမ်းဆည်းခြင်း (Scan အဆင့်)
Google Drive ၏ mime_type == 'application/vnd.google-apps.folder' ကို စစ်ဆေး၍ is_folder column သို့မဟုတ် mime_type ဖြင့် ခွဲခြားထည့်သွင်းပါ:

def insert_scanned_item(self, file_data: dict, account_id: int):
    """Scan ဖတ်ရရှိသော File/Folder များကို DB ထဲ ထည့်သွင်းခြင်း"""
    is_folder = 1 if file_data.get("mimeType") == "application/vnd.google-apps.folder" else 0
    parent_id = file_data.get("parents", [None])[0] if file_data.get("parents") else None

        file_data["id"],
        account_id,
        file_data["name"],
        file_data["mimeType"],
        parent_id,
        int(file_data.get("size", 0)),
        is_folder
    ))

def populate_download_queue(self, queue_id: int, account_id: int, download_path: str):
    # Folder မဟုတ်သော File များကိုသာ Query လုပ်၍ ဆွဲထုတ်မည်
    files = self.db.fetch_all("""
        SELECT id, name, mime_type, parent_id 
        FROM drive_files 
        WHERE account_id = ? AND is_folder = 0
    """, (account_id,))

    for file in files:
        # File ၏ Parent Folders များကို လိုက်ရှာပြီး Local Path တည်ဆောက်မည်
        rel_path = self._get_relative_file_path(file["name"], file.get("parent_id"), account_id)
        dest_file_path = os.path.join(download_path, rel_path)

        # download_items ထဲသို့ ထည့်သွင်းမည်
        self.db.execute("""
            INSERT INTO download_items (queue_id, drive_file_id, destination_path, status)
            VALUES (?, ?, ?, 'pending')
        """, (queue_id, file["id"], dest_file_path))
အဆင့် (၃) - Local Disk ပေါ်တွင် Folder လမ်းကြောင်းများ တည်ဆောက်ခြင်း (Download အဆင့်)
Worker က download_items ထဲမှ ဖိုင်ကို aria2c ဖြင့် မဒေါင်းမီ မိခင် Folder မရှိသေးပါက အလိုအလျောက် တည်ဆောက်ပေးရန် os.makedirs ကို သုံးပါ:

Python
# Download Worker ထဲတွင် ဖိုင်ဒေါင်းလုဒ် မစတင်မီ ခေါ်ရန်
file_directory = os.path.dirname(item["destination_path"])

# Local Path မရှိသေးပါက Sub-folders များကို အလိုအလျောက် ဆောက်ပေးမည်
os.makedirs(file_directory, exist_ok=True)

# ထို့နောက်မှ aria2c ဖြင့် ဒေါင်းလုဒ် စတင်မည်
run_aria2c_download(item["drive_file_id"], item["destination_path"])
လုပ်ငန်းစဉ် အနှစ်ချုပ်:

Scan လုပ်ချိန်: File နှင့် Folder နှစ်ခုလုံးကို drive_files ထဲ ထည့်ပါ။

Queue လုပ်ချိန်: is_folder = 0 ဖြစ်သော File များကိုသာ download_items ထဲ ကူးထည့်ပါ။

Download လုပ်ချိန်: os.makedirs() က လိုအပ်သော Folder များအား Local ပေါ်တွင် တည်ဆောက်ပေးပြီး File ကို ဒေါင်းလုဒ် ဆွဲပေးပါမည်


Google Drive ထဲက ဖိုင်အားလုံးကို ဒေါင်းလုဒ်ဆွဲသည့်အခါ "ဖိုင်များ Local ပေါ်တွင် ရှိပြီးသားဖြစ်နေခြင်း" နှင့် "Google Drive ပေါ်တွင် နာမည်တူ ဖိုင်များ ရှိနေခြင်း" စသည့် ပြဿနာများကို စနစ်တကျ ကိုင်တွယ်နိုင်ရန် အောက်ပါ Strategy (၃) မျိုး ကို ထည့်သွင်းပေးသင့်ပါသည်။

၁။ Local Disk ပေါ်တွင် ရှိပြီးသား ဖိုင်များကို စစ်ဆေး၍ Skip လုပ်ခြင်း (File Exist Check)
ဖိုင်မဒေါင်းမီ Local Disk ပေါ်တွင် ထိုဖိုင် ရှိပြီးသားလား နှင့် ဖိုင် Size တူမတူ စစ်ဆေးပါ။

ဖိုင်ရှိပြီး Size တူပါက: ဒေါင်းလုဒ် မဆွဲတော့ဘဲ Skip လုပ်မည်။

ဖိုင်ရှိသော်လည်း Size မတူပါက (မပြည့်စုံပါက): aria2c ၏ -c (Continue/Resume) flag ဖြင့် ဆက်လက် ဒေါင်းလုဒ်ဆွဲမည်။

၂။ Google Drive ပေါ်ရှိ နာမည်တူ ဖိုင်များအတွက် Auto Rename ပြုလုပ်ခြင်း
Google Drive တွင် Folder တစ်ခုတည်း၌ နာမည်တူ ဖိုင်အများအပြား ရှိနေနိုင်ပါသည်။ Local Disk ပေါ်တွင် တစ်ခုနှင့်တစ်ခု အစားထိုး (Overwrite) မသွားစေရန် နာမည်တူပါက filename (1).ext ပုံစံမျိုး အလိုအလျောက် နာမည်ပြောင်းပေးရန် လိုအပ်ပါသည်။

၃။ Database Level တွင် ပြီးပြီးသား ဖိုင်များကို မထည့်တော့ခြင်း
download_history ထဲတွင် status = 'completed' ဖြစ်ပြီးသား drive_file_id များကို Queue ထဲသို့ ထပ်မထည့်ရန် SQL Query တွင် ခွဲထုတ်ပေးပါ။

import os

def create_download_job_from_scanned_files(self, account_id: int, download_path: str):
    queue_id = self.download_mgr.create_queue(account_id=account_id, title="My Drive Sync Queue")
    
    # 1. download_history ထဲတွင် ပြီးပြီးသား ဖိုင်များကို ချန်လှပ်၍ ရွေးထုတ်မည်
    files = self.db.fetch_all("""
        SELECT f.id, f.name, f.mime_type, f.parent_id, f.size 
        FROM drive_files f
        WHERE f.account_id = ? 
          AND f.is_folder = 0
          AND f.id NOT IN (
              SELECT drive_file_id FROM download_history WHERE status = 'completed'
          )
    """, (account_id,))
    
    for file in files:
        rel_path = self._get_relative_file_path(file["name"], file.get("parent_id"), account_id)
        dest_file_path = os.path.join(download_path, rel_path)
        drive_file_size = file.get("size", 0)

        # 2. Local Disk ပေါ်တွင် ဖိုင်ရှိပြီးသားလား စစ်ဆေးမည်
        if os.path.exists(dest_file_path):
            local_file_size = os.path.getsize(dest_file_path)
            
            # ဖိုင်ရှိပြီး Drive က Size နဲ့ တူနေပါက မဒေါင်းဘဲ Skip လုပ်မည်
            if drive_file_size > 0 and local_file_size == drive_file_size:
                print(f"⏩ Skipped (Already downloaded): {file['name']}")
                continue

        # 3. Local Folder မရှိသေးပါက ဆောက်မည်
        os.makedirs(os.path.dirname(dest_file_path), exist_ok=True)
        
        # 4. Queue ထဲသို့ ထည့်သွင်းမည်
        self.download_mgr.add_item_to_queue(
            queue_id=queue_id,
            drive_file_id=file["id"],
            destination_path=dest_file_path,
            temp_path=dest_file_path + ".tmp"
        )

    self.download_mgr.start_queue(queue_id)
    return queue_id

Python
import os

def prepare_download_path(dest_file_path: str, drive_file_id: str) -> tuple[str, bool]:
    """
    ဖိုင်နာမည်တူနေပါက Conflict အဖြစ် မှတ်ပြီး ယာယီ Path ပြန်ပေးမည်
    """
    if not os.path.exists(dest_file_path):
        return dest_file_path, False # Duplicate မဟုတ်ပါ

    # နာမည်တူနေပါက Duplicate ဖြစ်သွားမည်
    dir_name, full_name = os.path.split(dest_file_path)
    file_name, ext = os.path.splitext(full_name)
    
    # ယာယီ နာမည်အဖြစ် Drive ID တွဲပေးမည် (ဥပမာ- report_1A2b3C.pdf)
    temp_name = f"{file_name}_{drive_file_id[:6]}{ext}"
    temp_path = os.path.join(dir_name, temp_name)
    
    return temp_path, True # Duplicate ဖြစ်ကြောင်း အချက်ပြမည်

# utils/helpers.py

def format_speed(bytes_per_sec: float) -> str:
    """Bytes/sec ကို KB/s သို့မဟုတ် MB/s သို့ ပြောင်းလဲပေးခြင်း"""
    if bytes_per_sec >= 1024 * 1024:
        return f"{bytes_per_sec / (1024 * 1024):.2f} MB/s"
    elif bytes_per_sec >= 1024:
        return f"{bytes_per_sec / 1024:.2f} KB/s"
    return f"{bytes_per_sec:.0f} B/s"


def calculate_eta(total_bytes: int, downloaded_bytes: int, speed_bytes_per_sec: float) -> str:
    """ကျန်ရှိသော အချိန် (Time Left) ကို HH:MM:SS သို့မဟုတ် MM:SS ပုံစံဖြင့် တွက်ပေးခြင်း"""
    remaining_bytes = total_bytes - downloaded_bytes
    
    if remaining_bytes <= 0:
        return "00:00"
    if speed_bytes_per_sec <= 0:
        return "Calculating..."

    seconds = int(remaining_bytes / speed_bytes_per_sec)
    hrs = seconds // 3600
    mins = (seconds % 3600) // 60
    secs = seconds % 60

    if hrs > 0:
        return f"{hrs:02d}:{mins:02d}:{secs:02d}"
    return f"{mins:02d}:{secs:02d}"

# core/download/download_manager.py
import time
from utils.helpers import format_speed, calculate_eta

class DownloadTracker:
    def __init__(self, total_bytes: int):
        self.total_bytes = total_bytes
        self.last_time = time.time()
        self.last_bytes = 0

    def calculate_metrics(self, current_bytes: int) -> tuple[str, str]:
        """
        ဒေါင်းလုဒ်ဆွဲပြီး ပမာဏကို ထည့်ပေးပါက Transfer Rate နှင့် Time Left ကို ပြန်ပေးမည်
        Returns: (speed_str, eta_str) -> ("5.20 MB/s", "02:15")
        """
        now = time.time()
        time_diff = now - self.last_time

        # 0.5 စက္ကန့်တိုင်း သို့မဟုတ် 1 စက္ကန့်တိုင်း Speed ကို ပြန်တွက်မည်
        if time_diff >= 0.5:
            bytes_diff = current_bytes - self.last_bytes
            speed_bps = bytes_diff / time_diff if time_diff > 0 else 0

            self.last_time = now
            self.last_bytes = current_bytes

            speed_str = format_speed(speed_bps)
            eta_str = calculate_eta(self.total_bytes, current_bytes, speed_bps)
            return speed_str, eta_str

        return "0 B/s", "--:--"

# core/download/queue_manager.py
from core.database.queue_repo import QueueRepository
from core.download.aria2_engine import Aria2Engine
from utils.logger import logger

class QueueManager:
    def __init__(self, db_manager, max_concurrent: int = 3):
        self.db = db_manager
        self.queue_repo = QueueRepository(db_manager)
        self.aria2 = Aria2Engine()
        self.max_concurrent = max_concurrent  # ပြိုင်တူဒေါင်းမည့် အရေအတွက် (Default = 3)

    def process_next_in_queue(self):
        """
        Queue ထဲမှ Priority အမြင့်ဆုံးဖိုင်များကို ရွေးထုတ်ပြီး 
        Max Concurrent Limit ထက် မကျော်စေဘဲ ဒေါင်းလုဒ် စတင်မောင်းနှင်မည်
        """
        # 1. လက်ရှိ Active ဖြစ်နေသော (Downloading) အရေအတွက်ကို စစ်မည်
        active_downloads = self.queue_repo.get_count_by_status("downloading")

        if active_downloads >= self.max_concurrent:
            logger.info(f"Max concurrent limit ({self.max_concurrent}) reached. Waiting...")
            return

        # 2. ကျန်ရှိနေသည့် slots အရေအတွက်ကို တွက်မည်
        available_slots = self.max_concurrent - active_downloads

        # 3. DB ထဲမှ Priority အမြင့်ဆုံး (10 > 5 > 1) pending ဖိုင်များကို ဆွဲထုတ်မည်
        # Query: SELECT * FROM download_items WHERE status = 'pending' ORDER BY priority DESC, id ASC LIMIT ?
        pending_items = self.queue_repo.get_pending_items_by_priority(limit=available_slots)

        for item in pending_items:
            self._start_download_item(item)

    def _start_download_item(self, item: dict):
        """Aria2 Engine သို့ Task လွှဲပြောင်း၍ ဒေါင်းလုဒ်စတင်ခြင်း"""
        item_id = item["id"]
        url = item["download_url"]
        save_path = item["save_path"]

        logger.info(f"Starting Download for Item #{item_id} (Priority: {item['priority']})")
        
        # Status ကို Downloading သို့ ပြောင်းမည်
        self.queue_repo.update_status(item_id, "downloading")
        
        # Aria2c ဖြင့် 16-Connections ခွဲ၍ စတင်ဆွဲမည်
        self.aria2.add_download(url=url, output_path=save_path, connections=16)

    def set_priority(self, download_item_id: int, priority_level: int):
        """
        User က သီးသန့် ဖိုင်တစ်ခုကို အရင်ဆုံး မြန်မြန်ဒေါင်းခိုင်းလိုပါက 
        Priority မြှင့်ပေးရန် (High: 10, Normal: 5, Low: 1)
        """
        self.queue_repo.update_item_priority(download_item_id, priority_level)
        logger.info(f"Item #{download_item_id} Priority set to {priority_level}")
        
        # Queue ကို ချက်ချင်း Re-check ပြုလုပ်မည်
        self.process_next_in_queue()

        "-x16",                          # Server သို့ Max Connections ၁၆ ခု ချိတ်မည်
        "-k1M",                          # 1MB ထက်ကြီးပါက အပိုင်း စတင်ခွဲမည်
        "--min-split-size=1M",
        "--file-allocation=falloc",      # Disk Space ကို ကြိုတင်နေရာယူမည် (Corruption ကာကွယ်ရန်)
        "--lowest-speed-limit=1K",       # Speed ကျသွားပါက Connection ပြန် Re-connect လုပ်မည်
        "--max-resume-failure-tries=0",  # Connection ကျပါက အကန့်အသတ်မရှိ ပြန်ချိတ်မည်
        "--retry-wait=2",                # Retry ပြုလုပ်ရန် စောင့်ဆိုင်းချိန် ၂ စက္ကန့်
        "--allow-overwrite=true"
    ]

    if speed_limit:
        cmd.append(f"--max-download-limit={speed_limit}")

    return cmd
ရရှိမည့် အကျိုးကျေးဇူး
Resume URL Refresh: Link သက်တမ်းကုန်သွားလည်း Downloader က 0% မှ ပြန်မစဘဲ ရပ်ထားသည့် ရာခိုင်နှုန်းမှအတိုင်း ဆက်လက်ဒေါင်းလုဒ် ဆွဲပေးပါမည်။

Queue Priority: ဖိုင် ၁၀၀ ဒေါင်းလုဒ်ထည့်ထားလည်း စက်မဟန်းဘဲ ၃ ဖိုင်စီ အလှည့်ကျ စနစ်တကျ ဒေါင်းပေးသွားပါမည်။

User များထံတွင် ဖြစ်ပေါ်သော Error များနှင့် Crash Log များကို မိမိထံသို့ အလိုအလျောက် သို့မဟုတ် တိုက်ရိုက် ရောက်ရှိလာစေရန် Error Reporting System တစ်ခု ချိတ်ဆက်ထားရန် လိုအပ်ပါသည်။

ယင်းအတွက် အသုံးပြုနိုင်သော အထိရောက်ဆုံး နည်းလမ်း (၃) ခုမှာ အောက်ပါအတိုင်း ဖြစ်ပါသည်။

# utils/error_reporter.py
import requests
import traceback
import platform

TELEGRAM_BOT_TOKEN = "YOUR_BOT_TOKEN"
TELEGRAM_CHAT_ID = "YOUR_CHAT_ID"

def send_error_to_telegram(error_exception):
    """Error တက်ပါက Telegram သို့ စာပို့ပေးမည့် Function"""
    try:
        error_trace = traceback.format_exc()
        system_info = f"OS: {platform.system()} {platform.release()}"
        
        message = (
            f"⚠️ **Nexora App Error Report**\n\n"
            f"**System:** {system_info}\n"
            f"**Error Details:**\n```\n{error_trace[:3000]}\n```" # စာလုံးရေ မပိုစေရန် 3000 ထိဖြတ်မည်
        )

        url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
        requests.post(url, json={"chat_id": TELEGRAM_CHAT_ID, "text": message, "parse_mode": "Markdown"})
    except Exception:
        pass # Network မရှိပါက Ignore လုပ်မည်

Python
# ui/main_window.py
import tkinter as tk
from tkinter import messagebox
from utils.error_reporter import send_error_to_telegram

class MainWindow(tk.Tk):
    def __init__(self):
        super().__init__()
        
        # Tkinter ၏ Global Exception Handler ကို ချိန်ညှိခြင်း
        self.report_callback_exception = self.show_error_dialog

    def show_error_dialog(self, exc, val, tb):
        """App တွင် Error တက်သွားပါက User ထံ Pop-up ပြပြီး Developer ထံ Log ပို့မည်"""
        error_msg = f"{exc.__name__}: {val}"
        
        # 1. Developer ထံ နောက်ကွယ်မှ Error Log လှမ်းပို့မည်
        send_error_to_telegram(val)

        # 2. User ထံ ယဉ်ကျေးစွာ အသိပေးမည်
        messagebox.showerror(
            "Unexpected Error", 
            f"An error occurred: {error_msg}\n\nThe issue has been logged and reported to the developer."
        )
